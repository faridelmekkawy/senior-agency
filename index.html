Data model (for your dev team)

Vendor-first style, but for multi-school uniforms. Everything is auto-created by the HTML apps.
agencies/{agencyId}
  - name
  - createdAt
  - bootstrapOwner (bool)
  - snapshotsEnabled (bool)

agencies/{agencyId}/schools/{schoolId}
  - name
  - active
  - createdAt
  - orderingOpen (bool)
  - sizeEditsUntil (timestamp)
  - sizeLockDate (timestamp)  // the date after which sizes become read-only
  - deadlines: [
      { key:"d1", dueAt, minPct:25 },
      { key:"d2", dueAt, minPct:50 },
      { key:"d3", dueAt, minPct:75 },
      { key:"d4", dueAt, minPct:100 }
    ]
  - percentSchedule: [25,25,25,25] // optional for UI
  - timeline: { commitmentDueAt, orderingDueAt, productionStartAt, deliveryDueAt } // optional

agencies/{agencyId}/schools/{schoolId}/students/{studentId}
  - fullName
  - phone
  - grade
  - committedAt (timestamp|null)
  - approvedAt (timestamp|null)
  - paidPct (number 0..100)
  - paymentStatus ("none"|"pending"|"approved"|"rejected")
  - orderedAt (timestamp|null)
  - sizesLocked (bool)
  - blocked (bool)
  - blockReason (string|null)
  - updatedAt

agencies/{agencyId}/schools/{schoolId}/snapshots/{YYYY-MM-DD}
  - commitmentPct, paidPct, orderedPct, lockedPct

agencies/{agencyId}/staff/{uid}
  - fullName
  - email
  - role ("owner"|"finance"|"ops"|"product"|"warehouse"|"viewer")
  - schools: ["schoolId1", ...] or ["*"] for all
  - active (bool)
  - createdAt

agencies/{agencyId}/audit/{autoId}
  - at, actorUid, actorName, action, payload
